\problem{1}
\begin{equation*}
    \hat i\times \hat j
	=\begin{vmatrix}
		\hat i & \hat j & \hat k \\
		\hat i_x & \hat i_x & \hat i_x \\
		\hat j_x & \hat j_x & \hat j_x \\
	\end{vmatrix}
	=\begin{vmatrix}
		\hat i & \hat j & \hat k \\
		1 & 0 & 0 \\
		0 & 1 & 0 \\
	\end{vmatrix}
	=
	\hat i \begin{vmatrix}
		0 & 0 \\ 1 & 0
	\end{vmatrix}
	-\hat j \begin{vmatrix}
		1 & 0 \\ 0 & 0
	\end{vmatrix}
	+\hat k \begin{vmatrix}
		1 & 0 \\ 0 & 1
	\end{vmatrix}
	=(0)\hat i-(0)\hat j+(1)\hat k
	=\boxed{\hat k=\left( 0,0,1 \right)}
\end{equation*}

	\hat j\times\hat i&=-\hat k &\qquad

	\hat k\times\hat i&=\hat j &\qquad
	\hat k\times\hat j&=-\hat i &\qquad
	\hat k\times\hat k&=0 \\
\end{alignat*}

\problem{3}
Cross product is anti-commutative: $a\times b=-b\times a$. 

\begin{proof}
	First, notice that the determinant of a matrix formed by swapping the rows of a $2\times 2$ matrix is the negative determinant of the original matrix:
	\begin{equation*}
		\begin{vmatrix}
			p & q \\
			r & s
		\end{vmatrix}
		=ps-qr
		=-(qr-ps)
		=-(rq-sp)
		=-\begin{vmatrix}
			r & s \\
			p & q
		\end{vmatrix}
	\end{equation*}
	Let $a,b$ be vectors in $\mathbb R^3$ with the components $a=(a_x,a_y,a_z)$ and $b=(b_x,b_y,b_z)$.
	\begin{align*}
	    a\times b
		&=\begin{vmatrix}
			\hat i & \hat j & \hat k \\
			a_x & a_y & a_z \\
			b_x & b_y & b_z \\
		\end{vmatrix}
		=
		\hat i \begin{vmatrix}
			a_y & a_z \\
			b_y & b_z
		\end{vmatrix}
		-\hat j \begin{vmatrix}
			a_x & a_z \\
			b_x & b_z
		\end{vmatrix}
		+\hat k \begin{vmatrix}
			a_x & a_y \\
			b_x & b_y
		\end{vmatrix} \\
	    -b\times a
		&=-\begin{vmatrix}
			\hat i & \hat j & \hat k \\
			b_x & b_y & b_z \\
			a_x & a_y & a_z \\
		\end{vmatrix}
		=-\left(
		\hat i \begin{vmatrix}
			b_y & b_z \\
			a_y & a_z
		\end{vmatrix}
		-\hat j \begin{vmatrix}
			b_x & b_z \\
			a_x & a_z
		\end{vmatrix}
		+\hat k \begin{vmatrix}
			b_x & b_y \\
			a_x & a_y
		\end{vmatrix}\right) \\
		&=\hat i \left(-\begin{vmatrix}
			b_y & b_z \\
			a_y & a_z
		\end{vmatrix}\right)
		-\hat j \left(-\begin{vmatrix}
			b_x & b_z \\
			a_x & a_z
		\end{vmatrix}\right)
		+\hat k \left(-\begin{vmatrix}
			b_x & b_y \\
			a_x & a_y
		\end{vmatrix}\right) \\
		\intertext{Using the observation earlier, to find the negative determinant, we can swap the rows of the matrix:}
		&=
		\hat i \begin{vmatrix}
			a_y & a_z \\
			b_y & b_z
		\end{vmatrix}
		-\hat j \begin{vmatrix}
			a_x & a_z \\
			b_x & b_z
		\end{vmatrix}
		+\hat k \begin{vmatrix}
			a_x & a_y \\
			b_x & b_y
		\end{vmatrix}
		=a \times b
	\end{align*}
\end{proof}
